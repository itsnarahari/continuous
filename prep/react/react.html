// src/features/auth/authSlice.js
import { createSlice } from '@reduxjs/toolkit';
import { loginUser, logoutUser, refreshToken, fetchMe, updateRolesOnServer, updateUserOnServer } from './authThunks';

/**
 * State:
 *  - token: JWT access token (string | null)
 *  - roles: string[]
 *  - user: { id, name, email, ... } | null
 *  - expiresAt: epoch millis (number | null)
 *  - status: 'idle' | 'loading' | 'succeeded' | 'failed'
 *  - error: string | null
 */
const initialState = {
  token: null,
  roles: [],
  user: null,
  expiresAt: null,
  status: 'idle',
  error: null,
};

const authSlice = createSlice({
  name: 'auth',
  initialState,
  reducers: {
    /** Add a role locally (idempotent) */
    addRole(state, action) {
      const role = action.payload;
      if (!state.roles.includes(role)) state.roles.push(role);
    },
    /** Remove a role locally */
    removeRole(state, action) {
      const role = action.payload;
      state.roles = state.roles.filter(r => r !== role);
    },
    /** Clear all roles */
    clearRoles(state) { state.roles = []; },
    /** Shallow-merge user fields (local) */
    updateUser(state, action) {
      state.user = { ...(state.user || {}), ...(action.payload || {}) };
    },
    /** Set all credentials in one shot (e.g., SSO callback) */
    setCredentials(state, action) {
      const { token, roles = [], user = null, expiresAt = null } = action.payload || {};
      state.token = token; state.roles = roles; state.user = user; state.expiresAt = expiresAt;
      state.error = null; state.status = 'succeeded';
    },
    /** Local-only logout (no API call). */
    localLogout(state) { Object.assign(state, initialState); },
  },
  extraReducers: (builder) => {
    builder
      // LOGIN
      .addCase(loginUser.pending, (s) => { s.status = 'loading'; s.error = null; })
      .addCase(loginUser.fulfilled, (s, a) => {
        const { token, roles, user, expiresAt } = a.payload;
        s.status = 'succeeded'; s.token = token; s.roles = roles || []; s.user = user || null; s.expiresAt = expiresAt || null;
      })
      .addCase(loginUser.rejected, (s, a) => { s.status = 'failed'; s.error = a.payload || a.error?.message || 'Login failed'; })

      // REFRESH
      .addCase(refreshToken.fulfilled, (s, a) => {
        const { token, expiresAt } = a.payload; s.token = token; s.expiresAt = expiresAt || null;
      })

      // /me
      .addCase(fetchMe.fulfilled, (s, a) => {
        const { user, roles } = a.payload; if (user) s.user = user; if (roles) s.roles = roles;
      })

      // server roles update
      .addCase(updateRolesOnServer.fulfilled, (s, a) => { s.roles = a.payload.roles; })

      // server user update
      .addCase(updateUserOnServer.fulfilled, (s, a) => { s.user = a.payload.user; })

      // LOGOUT (server)
      .addCase(logoutUser.fulfilled, (s) => { Object.assign(s, initialState); });
  }
});

export const { addRole, removeRole, clearRoles, updateUser, setCredentials, localLogout } = authSlice.actions;
export default authSlice.reducer;

// src/lib/apiClient.js
/**
 * Minimal fetch wrapper: adds Authorization header if token provided.
 * You can replace with axios; this keeps example simple & dependency-free.
 */
export async function apiFetch(path, { method = 'GET', token, body, headers = {}, ...rest } = {}) {
  const h = { 'Content-Type': 'application/json', ...headers };
  if (token) h.Authorization = `Bearer ${token}`;
  const res = await fetch(path, {
    method,
    headers: h,
    body: body ? JSON.stringify(body) : undefined,
    ...rest,
  });
  // Normalize errors
  if (!res.ok) {
    const text = await res.text().catch(() => '');
    throw new Error(text || `HTTP ${res.status}`);
  }
  // Allow empty bodies
  const ct = res.headers.get('content-type') || '';
  return ct.includes('application/json') ? res.json() : null;
}

// src/features/auth/authThunks.js
import { createAsyncThunk } from '@reduxjs/toolkit';
import { apiFetch } from '../../lib/apiClient';
import { selectToken } from './authSelectors';

/** Helper to compute expiry from JWT "exp" claim if server doesn't send */
function expiresFromJwt(token) {
  try {
    const [, payloadB64] = token.split('.');
    const payload = JSON.parse(atob(payloadB64.replace(/-/g,'+').replace(/_/g,'/')));
    return payload.exp ? payload.exp * 1000 : null;
  } catch { return null; }
}

/** Login: POST /auth/login {email, password} -> {token, roles, user, expiresAt?} */
export const loginUser = createAsyncThunk(
  'auth/login',
  async (creds, { rejectWithValue }) => {
    try {
      const data = await apiFetch('/auth/login', { method: 'POST', body: creds });
      // normalize
      const token = data.token;
      const expiresAt = data.expiresAt ?? (token ? expiresFromJwt(token) : null);
      return { token, roles: data.roles || [], user: data.user || null, expiresAt };
    } catch (e) { return rejectWithValue(e.message); }
  }
);

/** Logout: POST /auth/logout (optional). Always clear locally on success. */
export const logoutUser = createAsyncThunk(
  'auth/logout',
  async (_, { getState }) => {
    const token = selectToken(getState());
    try { await apiFetch('/auth/logout', { method: 'POST', token }); } catch { /* ignore */ }
    return true;
  }
);

/** Refresh: POST /auth/refresh -> {token, expiresAt?} */
export const refreshToken = createAsyncThunk(
  'auth/refresh',
  async (_, { getState, rejectWithValue }) => {
    const token = selectToken(getState());
    try {
      const data = await apiFetch('/auth/refresh', { method: 'POST', token });
      const next = data.token;
      const expiresAt = data.expiresAt ?? (next ? expiresFromJwt(next) : null);
      return { token: next, expiresAt };
    } catch (e) { return rejectWithValue(e.message); }
  }
);

/** /me: GET current user & roles */
export const fetchMe = createAsyncThunk(
  'auth/me',
  async (_, { getState, rejectWithValue }) => {
    const token = selectToken(getState());
    try { return await apiFetch('/auth/me', { token }); }
    catch (e) { return rejectWithValue(e.message); }
  }
);

/** Update roles on server (PUT /users/:id/roles) */
export const updateRolesOnServer = createAsyncThunk(
  'auth/updateRoles',
  async ({ userId, roles }, { getState, rejectWithValue }) => {
    const token = selectToken(getState());
    try { 
      const data = await apiFetch(`/users/${userId}/roles`, { method: 'PUT', token, body: { roles } });
      return { roles: data.roles ?? roles };
    } catch (e) { return rejectWithValue(e.message); }
  }
);

/** Update user details on server (PATCH /users/:id) */
export const updateUserOnServer = createAsyncThunk(
  'auth/updateUser',
  async ({ userId, patch }, { getState, rejectWithValue }) => {
    const token = selectToken(getState());
    try { 
      const data = await apiFetch(`/users/${userId}`, { method: 'PATCH', token, body: patch });
      return { user: data.user ?? { id: userId, ...patch } };
    } catch (e) { return rejectWithValue(e.message); }
  }
);

  • logoutUser() (thunk) → POST /auth/logout (optional), clears local state.
  • refreshToken() (thunk) → POST /auth/refresh, updates token, expiresAt.
  • fetchMe() (thunk) → GET /auth/me, refreshes user/roles from server.

  • updateUserOnServer({userId, patch}) (thunk) → PATCH /users/:id, then state.user = server.

  • localLogout() (reducer) → clears local auth without calling server.

  • If you must persist, use redux-persist but understand the risk; rotate tokens frequently.

📌 1. What is the Virtual DOM?

The Virtual DOM (VDOM) is a lightweight, in-memory representation of the real DOM. React uses it to improve performance by reducing direct DOM manipulation.

It's like a blueprint of the actual UI elements.

📌 2. Why is Virtual DOM Needed?
  • ⚡ Updating the real DOM is slow (it's an expensive operation)
  • 🧠 Virtual DOM allows React to batch updates efficiently
  • 🎯 It minimizes the number of changes made to the real DOM
📌 3. How It Works (Step-by-Step)
  1. React renders components to the Virtual DOM
  2. It compares the new Virtual DOM with the previous one (using diffing algorithm)
  3. It calculates the minimum set of changes needed
  4. It efficiently updates the real DOM in one go (called reconciliation)
📌 4. Code Example

Consider this simple React component:


function Counter() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Count is: {count}</p>
      <button onClick={() => setCount(count + 1)}>Increment</button>
    </div>
  );
}

When you click the button, only the count text (`<p>`) is updated in the real DOM — not the whole UI.

📌 Conclusion

The Virtual DOM is a key reason why React is fast and scalable. It allows efficient UI updates by avoiding direct manipulation of the real DOM and only applying the necessary changes after diffing. It brings a smart and declarative way to build user interfaces.

Short answer: Both React and Vue can be very fast when used correctly. In many real apps the difference is negligible compared to how you code (state placement, memoization, virtualization, bundling).

How their performance models differ
  • Vue 3: fine-grained reactivity (Proxies) tracks exact dependencies → fewer components re-render by default. Templates are compiled to optimized render functions.
  • React 18: re-renders a component when its parent renders/props change; you opt in to skipping work using React.memo, useMemo, useCallback. Concurrent rendering improves responsiveness under load but doesn’t make pure CPU work “faster”.

Rule of thumb: out-of-the-box, Vue may do less work for small reactive updates; with good memoization, React matches it.

What actually makes apps “feel fast”
  • Critical path: bundle size, network waterfalls, SSR/SSG, hydration.
  • Work avoided: limiting renders, virtualizing large lists, splitting routes/components.
  • Scheduling: breaking long tasks, deferring non-critical work, using transitions.
  • Data strategy: colocate state; minimize global/state explosions and prop drilling.

Measure, don’t guess
  • Browser: Performance panel (FPS, long tasks); Lighthouse for LCP/INP/CLS.
  • React: React Profiler (record commits, “why did this render”).
  • Vue: Vue Devtools perf tab (component render times, reactive events).
  • Production: web-vitals, Real User Monitoring (RUM), flamecharts.
When one may edge the other
  • Many tiny reactive updates: Vue often wins with less tuning.
  • Heavy custom memoization/large trees: React performs great if you use memo + stable refs.
  • SSR/streaming: Both offer excellent SSR; pipeline and caching usually dominate differences.
Bottom line
  • Pick based on team expertise, ecosystem, and DX. With best practices, both deliver 60fps UIs.
  • Focus on architecture (state design, fetching, code-splitting, virtualization) more than framework choice.

Vite in one line: a next-gen frontend toolchain that gives instant dev server starts and blazing HMR using native ES modules, with a production build powered by Rollup.

How Vite works (under the hood)
  • Dev server: serves source files as native ESM. No heavy upfront bundling → near-instant cold start.
  • Pre-bundling: uses esbuild to prebundle deps once (super fast, Go-based), then caches them.
  • HMR: tracks a module graph; on save, it invalidates only affected modules and pushes updates via WebSocket.
  • Build: for production it hands off to Rollup → tree-shaking, code-splitting, optimized chunks.
Why teams integrate Vite with React
  • Speed & DX: sub-second cold starts and fast refresh even in large repos; low RAM compared to legacy Webpack setups.
  • React Fast Refresh: first-class via @vitejs/plugin-react or the faster @vitejs/plugin-react-swc.
  • Sane defaults: TS, JSX/TSX, CSS Modules, PostCSS, env files, and absolute import aliases with minimal config.
  • Modern standards: native ESM, dynamic imports, top-tier HMR; great for monorepos and component libraries.
  • Easier builds: Rollup output is predictable; easy code-splitting and library mode.

Short answer: Yes. Vue 2 and Vue 3 both use a Virtual DOM (VDOM). Vue 3 still diffs VDOM trees, but combines that with fine-grained reactivity so only components whose reactive data changed re-render, and the compiler adds hints to minimize the amount of diffing work.

How it works in Vue
  • Template → render function → VNodes: Vue compiles your SFC template into a render function that produces a tree of virtual nodes (VNodes).
  • Diff & patch: On state changes, Vue diffs the new VNode tree against the previous one and patches only the necessary parts of the real DOM.
  • Reactivity gates work: Vue tracks which reactive properties each component uses; when one changes, only those components re-render (others are skipped).
  • Compiler hints (Vue 3): the compiler hoists static nodes and adds patch flags so the runtime can skip whole subtrees or focus diffs (e.g., “only text changed”).

Why a Virtual DOM?
  • Performance: batch and minimize DOM mutations; skip static subtrees.
  • Portability: same render logic works for SSR, Native (via render targets), etc.
  • Predictability: declarative view updated from state; no manual DOM diffing.
Vue vs React (VDOM behavior)
  • Both use VDOM diff/patch.
  • Vue 3 also uses dependency tracking: a component won’t re-render if none of the reactive values it used changed. The compiler further narrows diffs with patch flags and static hoisting.
  • React re-renders from parents downward; you opt-out of work via React.memo/useMemo/useCallback.
Keys & list updates
  • Always provide a stable :key in v-for to preserve element identity and avoid DOM churn.
<li v-for="u in users" :key="u.id">{{ u.name }}</li>
When Vue “skips” work
  • v-once for static subtrees (render once, never update).
  • Compiler hoists static nodes out of the render function.
  • Patch flags let Vue update just text/props instead of re-diffing entire branches.
Common pitfalls
  • Missing/unstable :key in lists → unnecessary re-mounts.
  • Overly global state causing avoidable updates; keep state close to where it’s used.
  • Huge lists: VDOM still does work—use virtualization (e.g., vue-virtual-scroller).

📌 1. What is a Lifecycle in React?

The React lifecycle represents the different stages a component goes through from creation to destruction. You can tap into these stages to run your code (API calls, cleanup, etc.).

📌 2. Three Main Phases

  • Updating – State or props change, triggering re-render
  • Unmounting – Component is removed from DOM

    console.log("Component updated");
  }

    console.log("Component will unmount");
  }

  render() {

📌 1. What are Hooks?

Hooks are special functions introduced in React 16.8 that let you use React features (like state, lifecycle, context) in functional components.

📌 2. Basic Hooks
  • useState – Add local state
  • useEffect – Side effects like API calls, subscriptions
  • useContext – Access context without wrapper
📌 3. Additional Hooks
  • useReducer – Complex state management (like Redux)
  • useCallback – Memoizes a function between renders
  • useMemo – Memoizes a calculated value
  • useRef – Mutable reference that doesn’t cause re-render
  • useImperativeHandle – Customize the instance exposed by ref
  • useLayoutEffect – Like useEffect, but fires before paint
  • useDebugValue – Debugging custom hooks
  • useTransition – Optimize state transitions (React 18+)
  • useDeferredValue – Defer re-renders for expensive updates
  • useId – Generate unique IDs (React 18+)
  • useSyncExternalStore – Subscribe to external store safely
  • useInsertionEffect – Inject styles before layout (rare)
📌 4. When to Create a Custom Hook?
  • ♻️ Repeating same logic in multiple components
  • 🧼 Want to keep components clean and readable
  • 🔌 Need to isolate and reuse side effects
📌 5. How to Create a Custom Hook

Custom hooks are just JavaScript functions that start with use and can use other hooks inside.


// useWindowWidth.js
import { useState, useEffect } from 'react';

function useWindowWidth() {
  const [width, setWidth] = useState(window.innerWidth);

  useEffect(() => {
    const handleResize = () => setWidth(window.innerWidth);
    window.addEventListener('resize', handleResize);

    return () => window.removeEventListener('resize', handleResize);
  }, []);

  return width;
}

export default useWindowWidth;
📌 6. Using Custom Hook

  const width = useWindowWidth();
  return <p>Window width: {width}px</p>;
}
📌 7. Rules of Hooks
  • ✅ Always call hooks at the top level (not inside loops/conditions)
  • ✅ Only call hooks from React functions (or custom hooks)
  • ✅ Custom hook names must start with use
📌 Conclusion

React provides a rich set of built-in hooks for most needs. When logic repeats across components, building a custom hook makes code cleaner and reusable. All hooks follow strict rules to ensure predictable behavior.

📌 1. What is useMemo?

useMemo is a React Hook that caches a computed value so it doesn't have to be recalculated on every render.


const expensiveValue = useMemo(() => {
  return computeHeavyOperation(data);
}, [data]);

✅ Useful for performance optimization when computations are heavy or complex.

📌 2. What is useCallback?

useCallback returns a memoized version of a callback function that only changes if one of the dependencies changes.


const handleClick = useCallback(() => {
  console.log('Clicked!');
}, []);

✅ Useful when passing callbacks to child components to avoid unnecessary re-renders.

📌 3. Key Difference
  • useMemo – Caches the result of a function
  • useCallback – Caches the function reference itself
📌 4. Side-by-Side Example

const memoizedValue = useMemo(() => expensiveFunction(a, b), [a, b]);

const memoizedCallback = useCallback(() => {
  doSomething(a, b);
}, [a, b]);
📌 5. Real Use Case Example

function Parent() {
  const [count, setCount] = useState(0);
  const [text, setText] = useState('');

  const expensiveCalculation = useMemo(() => {
    console.log("Calculating...");
    return count * 2;
  }, [count]);

  const handleChange = useCallback((e) => {
    setText(e.target.value);
  }, []);

  return (
    <div>
      <h4>Double Count: {expensiveCalculation}</h4>
      <input onChange={handleChange} value={text} />
    </div>
  );
}
📌 6. When to Use?
  • 🧠 Use useMemo when:
    • Heavy calculations
    • Prevent re-computing values unnecessarily
  • 🧠 Use useCallback when:
    • Passing functions to child components
    • Prevent function recreation on every render
📌 7. Visual Summary

useCallback Function reference Prevent re-rendering children
📌 Conclusion

Both useMemo and useCallback are performance optimization hooks. Use them only when necessary, such as for expensive computations or preventing unnecessary re-renders of child components that rely on stable function references.

📌 1. What is React Fiber?

React Fiber is the **complete rewrite of React's core algorithm**, introduced in React 16. It enables **incremental rendering**, **better performance**, and **prioritization** of UI updates. Unlike the older stack-based algorithm, Fiber uses a **linked list structure** for better control over work units.

📌 2. Goals of React Fiber
  • 🔁 Incremental rendering (split rendering work into chunks)
  • 🎯 Prioritize updates based on urgency (like animations vs. data fetch)
  • ⏸️ Pause, resume, and restart work as needed
  • 🔄 Concurrency and smooth rendering under load
📌 3. How Fiber Works Internally

Fiber breaks down rendering into a tree of units of work. It uses a **work loop** to perform updates incrementally, allowing the browser to do other work in between (like user input or animations).


function workLoop(deadline) {
  while (nextUnitOfWork && deadline.timeRemaining() > 1) {
    nextUnitOfWork = performUnitOfWork(nextUnitOfWork);
  }
  if (!nextUnitOfWork && workInProgressRoot) {
    commitRoot();
  }
}

This allows React to split work across multiple frames for smooth rendering.

📌 4. Fiber Node Structure

Each Fiber node represents a component and contains information like:

  • type – Component type
  • stateNode – DOM node or class instance
  • child, sibling, return – For traversing the tree
  • pendingProps, memoizedProps – For props comparison
📌 5. Phases of Fiber

React Fiber has two main phases:

  1. Render Phase: Work is done on a copy of the Fiber tree (non-blocking)
  2. Commit Phase: Final DOM updates are committed (blocking)
📌 6. Reconciliation with Fiber

During reconciliation, React compares new elements with old ones and determines what to update, remove, or add. Fiber allows this to be done **asynchronously**, improving performance for large trees.

📌 7. Scheduling and Priority

React Fiber uses an internal priority system based on **"lanes"**, where:

  • High-priority tasks like input get processed first
  • Low-priority tasks (like logging) can be deferred
This makes React highly responsive under load.

📌 8. Concurrency with Fiber

Fiber is the foundation for **Concurrent Mode** in React. It enables:

  • Suspense for data fetching
  • Time-slicing (pausing work mid-way)
  • Better responsiveness for high-priority updates

📌 9. Real-World Benefits

With Fiber, React apps:

  • Feel smoother with animations
  • Don't block user interactions while rendering
  • Handle complex UIs better

📌 Conclusion

React Fiber is a powerful re-architecture of the core React rendering engine. It improves responsiveness, enables concurrency, and lays the groundwork for modern React features like Suspense and Concurrent Rendering. It’s a core part of what makes React performant today.

📌 What is Concurrent Mode?

Concurrent Mode is an advanced feature in React that helps make your app more responsive by rendering components *without blocking the main thread*. It allows React to pause, abort, or restart rendering work as needed.

📌 Key Benefits
  • 🚀 Improves performance for large, complex UIs
  • 💡 Enables better user experience by avoiding blocking interactions
  • 🔁 Supports features like **Suspense**, **Transitions**, **startTransition**, etc.
📌 Real World Example

In heavy UI apps (like search filters), React can skip unnecessary renders and prioritize important updates like input typing.

📌 What is Reconciliation?

Reconciliation is the process by which React updates the **DOM** when the **state or props** of a component change. It compares the new Virtual DOM tree with the previous one and applies the minimal number of changes to the real DOM.

📌 How It Works
  • 🧠 React maintains a Virtual DOM
  • 📤 When state/props change, a new Virtual DOM is created
  • 🕵️‍♂️ React compares the new Virtual DOM with the old one
  • 🔧 Only the changed elements are updated in the real DOM
📌 Key Algorithm: Diffing

React uses a **diffing algorithm** with these assumptions:

  • 📦 Elements of different types produce different trees
  • 🔁 You can hint React with a key prop to optimize reconciliation for lists
📌 Example

// Initial render
<ul>
  <li key="1">A</li>
  <li key="2">B</li>
</ul>

// After update
<ul>
  <li key="2">B</li>
  <li key="3">C</li>
</ul>

🔑 Using key helps React move/update elements correctly without re-rendering the whole list.

📌 Optimizations React Uses
  • 🧬 Component-level re-rendering only if props/state change
  • 🔁 Skips subtrees if component returns same result
  • 🚀 React.memo and PureComponent help skip unnecessary renders
📌 Real-World Tip

Always use key for list items. Avoid using index as key unless list never changes.

📌 Conclusion

Reconciliation makes React efficient by updating only the parts of the DOM that changed. This allows for fast and scalable UI updates even with complex components.

Modern React favors functional components with Hooks for most use-cases. Class components remain for legacy code and Error Boundaries.

📌 Conceptual Differences
  • Syntax & mental model: Classes use this, lifecycle methods, and setState. Functions use plain JS functions and Hooks.
  • State: Class → this.state + this.setState(). Function → useState, useReducer.
  • Lifecycle: Class → componentDidMount/DidUpdate/WillUnmount. Function → useEffect/useLayoutEffect.
  • Reusability: Classes rely on HOCs/render props; Functions use custom hooks (cleaner composition).
  • Error boundaries: As of now, only class components can define error boundaries (componentDidCatch/getDerivedStateFromError).
  • Performance: Functional components are leaner; hooks like useMemo/useCallback help control re-renders.
🧩 Side-by-side Example
// Class component
class Counter extends React.Component {
  state = { count: 0 };
  componentDidMount() { document.title = `Count: ${this.state.count}`; }
  componentDidUpdate() { document.title = `Count: ${this.state.count}`; }
  componentWillUnmount() { /* cleanup */ }
  render() {
    return <button onClick={() => this.setState({count: this.state.count + 1})}>
      {this.state.count}
    </button>;
  }
}
// Functional component with Hooks
function Counter() {
  const [count, setCount] = React.useState(0);
  React.useEffect(() => {
    document.title = `Count: ${count}`;
    return () => { /* cleanup */ };
  }, [count]);
  return <button onClick={() => setCount(c => c + 1)}>{count}</button>;
}

  • New components: default to functional + hooks.
  • Logic reuse: extract to custom hooks (useAuth, useFetch, etc.) instead of HOCs.
  • Complex state: prefer useReducer over deeply nested useState.
  • Refs & instance vars: use useRef for mutable values that don’t trigger re-renders.
  • Performance: use React.memo, useMemo, useCallback to avoid unnecessary renders on hot paths.
🏷️ When to Keep/Use Class Components
  • Error boundaries that catch descendant render errors (wrap your functional tree with a class boundary).
  • Legacy codebases where migration risk is high—introduce hooks gradually for new parts.

React builds a lightweight virtual tree of UI (elements) and compares it to the previous one. It computes the minimal set of changes and applies them to the real DOM efficiently. Since React 16+, this is implemented by the Fiber architecture.

🧠 Mental Model
  • Your component returns elements (plain objects describing UI).
  • React creates a Fiber tree (one Fiber per element) to track work, props, state, lanes/priority.
  • On state/prop change, React renders a new tree and diffs it with the previous tree → produces mutations.
  • Mutations are applied in the commit phase (DOM updates, refs, lifecycle/effects).
🧵 Two Phases (Fiber)
  • Render phase (reconciliation): can be interrupted/paused; computes a new Fiber tree and a list of DOM ops.
  • Commit phase: synchronous; applies DOM updates, runs useLayoutEffect/componentDidMount/Update, then schedules useEffect.
🔍 Diffing Rules (Reconciliation)
  • Same type element (e.g., <button> → <button>): update props; keep DOM node; update children.
  • Different type (<button> → <div>): replace DOM node; unmount old subtree; mount new.
  • Lists use keys: keys identify item identity across renders. React matches by key first, then by index if missing.
// Good keys: stable IDs from data
items.map(item => <Row key={item.id} data={item} />)

// Bad: index as key (causes wrong reuse on insert/remove)
items.map((item, i) => <Row key={i} data={item} />)
🧩 Key Effects (Mount vs Update)
// Reordering with stable keys → updates/moves DOM nodes, state preserved per key
[ {id:1}, {id:2} ]  → [ {id:2}, {id:1} ]

// Changing a component's key forces remount (state reset)
<Form key={userId} /> // new userId ⇒ new Form state
⚡ Batching & Priorities (Concurrent React)
  • Multiple state updates in the same tick are batched to reduce renders.
  • Concurrent features assign priorities (lanes): urgent updates (typing/clicks) vs. transitions (non-urgent UI).
// Deprioritize expensive UI updates
const [isPending, startTransition] = React.useTransition();
startTransition(() => setSearchQuery(q)); // can be interrupted/yielded
🧭 When Does React Re-render?
  • Parent re-renders → children re-render by default (because new props).
  • React.memo(Component) skips re-render if props are shallow-equal.
  • useMemo/useCallback stabilize values/functions passed to children.
const Child = React.memo(function Child({onClick, data}) { /* ... */ });

const onClick = React.useCallback(() => doThing(id), [id]);
const expensive = React.useMemo(() => compute(data), [data]);
<Child onClick={onClick} data={expensive} />

🧯 Effects Order & Lifecycles
  • useLayoutEffect: runs synchronously after DOM mutations, before paint (measure/layout).
  • useEffect: runs after paint (non-blocking side effects, data fetching, subscriptions).
  • On unmount or key change, cleanup functions run before the node is removed/replaced.
⚠️ Common Pitfalls
  • Using index as key in dynamic lists → visual glitches, wrong state association.
  • Creating new objects/functions every render passed to memoized children → defeats memoization.
  • Heavy work in render/effects → jank; memoize or move off main thread (Web Worker) when needed.
  • Accidentally changing a component’s key on every render → constant remounts.
✅ Practical Tips
  • Always use stable keys from domain IDs.
  • Wrap pure presentational children with React.memo; pass stable props via useMemo/useCallback.
  • For large lists, use windowing (react-window/react-virtualized).
  • Use transitions (useTransition/useDeferredValue) to keep typing responsive during expensive updates.

Bottom line: Reconciliation compares element trees (via keys/type) to compute minimal DOM mutations. Fiber lets React schedule, pause, and prioritize work so your UI stays responsive.

// Context + useReducer pattern (split state & dispatch to reduce re-renders)
import React from "react";

const CartStateCtx = React.createContext(null);
const CartDispatchCtx = React.createContext(null);

function cartReducer(state, action) {
  switch (action.type) {
    case "add": {
      const { id } = action.item;
      const qty = (state.items[id]?.qty || 0) + 1;
      return { ...state, items: { ...state.items, [id]: { ...action.item, qty } } };
    }
    case "remove": {
      const s = { ...state.items }; delete s[action.id];
      return { ...state, items: s };
    }
    default: return state;
  }
}

export function CartProvider({ children }) {
  const [state, dispatch] = React.useReducer(cartReducer, { items: {} });
  // Memoizing only the state object reference won't block re-render of all consumers; split contexts does.
  return (
    <CartDispatchCtx.Provider value={dispatch}>
      <CartStateCtx.Provider value={state}>{children}</CartStateCtx.Provider>
    </CartDispatchCtx.Provider>
  );
}

export function useCart() {
  const ctx = React.useContext(CartStateCtx);
  if (!ctx) throw new Error("useCart must be inside CartProvider");
  return ctx;
}
export function useCartDispatch() {
  const ctx = React.useContext(CartDispatchCtx);
  if (!ctx) throw new Error("useCartDispatch must be inside CartProvider");
  return ctx;
}

// Usage
function AddToCartButton({ item }) {
  const dispatch = useCartDispatch();
  return <button onClick={() => dispatch({ type: "add", item })}>Add</button>;
}

function CartBadge() {
  const { items } = useCart();
  const count = React.useMemo(() => Object.values(items).reduce((a, b) => a + b.qty, 0), [items]);
  return <span>{count}</span>;
}

import { configureStore, createSlice, createSelector } from "@reduxjs/toolkit";
import { Provider, useDispatch, useSelector, shallowEqual } from "react-redux";

const cartSlice = createSlice({
  name: "cart",
  initialState: { items: {} },
  reducers: {
    add(state, action) {
      const { id } = action.payload;
      const curr = state.items[id];
      state.items[id] = { ...action.payload, qty: (curr?.qty || 0) + 1 }; // Immer lets you "mutate"
    },
    remove(state, action) { delete state.items[action.payload]; }
  }
});
export const { add, remove } = cartSlice.actions;

export const store = configureStore({ reducer: { cart: cartSlice.reducer } });

// Memoized selector (recomputes only when inputs change)
const selectItems = (state) => state.cart.items;
export const selectCount = createSelector([selectItems], (items) =>
  Object.values(items).reduce((a, b) => a + b.qty, 0)
);

}

  • Server state changes often, has caching, background refresh, and stale-while-revalidate needs.
  • Use TanStack Query (React Query) or RTK Query to manage it; keep Redux/Context for client-only state (UI prefs, wizard step, cart before checkout).

  • Using Context as a single “god provider” → every change re-renders the app.

✅ Decision guide
  • Tiny/medium app, few globals → Context + useReducer, split by domain.

function useRequest(url, opts = {}) {
  const [data, setData] = React.useState(null);
  const [error, setError] = React.useState(null);
  const [loading, setLoading] = React.useState(false);
  const [version, bump] = React.useState(0); // to force refetch

  const refetch = React.useCallback(() => bump(v => v + 1), []);
  React.useEffect(() => {
    const ctrl = new AbortController();
    let alive = true;
    (async () => {
      try {
        setLoading(true); setError(null);
        const res = await fetch(url, { ...opts, signal: ctrl.signal });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = await res.json();
        if (alive) setData(json);
      } catch (e) {
        if (e.name !== "AbortError" && alive) setError(e.message);
      } finally {
        if (alive) setLoading(false);
      }
    })();
    return () => { alive = false; ctrl.abort(); };
  }, [url, JSON.stringify(opts), version]); // keep opts stable in caller ideally

  return { data, error, loading, refetch };
}

4) Retry with exponential backoff (transient failures)
async function fetchWithRetry(input, init, { retries = 3, base = 300 } = {}) {
  let last;
  for (let i = 0; i <= retries; i++) {
    try {
      const res = await fetch(input, init);
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      return res;
    } catch (e) {
      last = e;
      if (i === retries) break;
      await new Promise(r => setTimeout(r, base * 2 ** i + Math.random() * 100));
    }
  }
  throw last;
}

Bottom line: Design for failure as the norm. Redundancy + isolation + automated failover keep you available; timeouts/retries/circuits & graceful degradation keep you useful; DR drills ensure you can recover when the worst happens.
