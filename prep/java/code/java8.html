import java.time.LocalDate;

class Employee {
  private final int id;
  private final String name;
  private final String gender;      // "Male"/"Female"
  private final int age;
  private final String department;
  private final double salary;
  private final LocalDate joinDate;
  private final List<String> hobbies;

  Employee(int id, String name, String gender, int age, String department, double salary, LocalDate joinDate, List<String> hobbies) {
    this.id = id; this.name = name; this.gender = gender;
    this.age = age; this.department = department; this.salary = salary;
    this.joinDate = joinDate; this.hobbies = hobbies == null ? Collections.emptyList() : hobbies;
  }
  public int getId(){ return id; }
  public String getName(){ return name; }
  public String getGender(){ return gender; }
  public int getAge(){ return age; }
  public String getDepartment(){ return department; }
  public double getSalary(){ return salary; }
  public LocalDate getJoinDate(){ return joinDate; }
  public List<String> getHobbies(){ return hobbies; }

  @Override public String toString(){
    return String.format("Emp{id=%d, name=%s, dept=%s, sal=%.2f}", id,name,department,salary);
  }
}

class Order {
  private final String customer;
  private final double amount;
  Order(String customer, double amount){ this.customer = customer; this.amount = amount; }
  public String getCustomer(){ return customer; }
  public double getAmount(){ return amount; }
}

String s = "ababa";
Set<String> palSubs =
  IntStream.range(0, s.length())
    .boxed()
    .flatMap(i -> IntStream.rangeClosed(i+1, s.length())
        .mapToObj(j -> s.substring(i,j)))
    .filter(sub -> sub.length() >= 2)
    .filter(sub -> new StringBuilder(sub).reverse().toString().equals(sub))
    .collect(Collectors.toCollection(LinkedHashSet::new));
System.out.println(palSubs);

  int y = target - x;
  if(seen.contains(y)){
    int min = Math.min(x,y), max = Math.max(x,y);
    pairs.add(min + "," + max);
  }
  seen.add(x);
}
